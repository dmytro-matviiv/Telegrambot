# Приклад файлу з проксі серверами
# Формати:
# ip:port
# ip:port:username:password
# http://ip:port
# http://username:password@ip:port

# Безкоштовні проксі (замініть на реальні)
# 8.8.8.8:8080
# 1.1.1.1:3128
# 192.168.1.1:8080

# Платні проксі (приклад)
# http://user:pass@proxy1.example.com:8080
# http://user:pass@proxy2.example.com:3128

# SOCKS проксі
# socks5://user:pass@proxy.example.com:1080

# Примітка: Для реального накручування потрібні робочі проксі сервери
# Рекомендується використовувати платні проксі для кращої стабільності
